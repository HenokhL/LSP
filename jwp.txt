============================================
  PRIME JACKET STORE - LARAVEL SETUP GUIDE
============================================

1. REQUIREMENT
--------------
- PHP 8.2+
- Composer
- XAMPP (MySQL + Apache)
- Node.js (opsional)

2. INSTALASI AWAL
-----------------
cd C:\xampp\htdocs\LSP
composer install
composer require barryvdh/laravel-dompdf

3. KONFIGURASI DATABASE
-----------------------
- Buka XAMPP Control Panel
- Start Apache dan MySQL
- Buka phpMyAdmin: http://localhost/phpmyadmin
- Buat database: primejacketstore
- Import file: primejacketstore.sql

4. KONFIGURASI .ENV
-------------------
DB_CONNECTION=mysql
DB_HOST=127.0.0.1
DB_PORT=3306
DB_DATABASE=primejacketstore
DB_USERNAME=root
DB_PASSWORD=

5. COMMAND WAJIB SEBELUM MENJALANKAN
-------------------------------------
composer dump-autoload
php artisan key:generate
php artisan config:clear
php artisan cache:clear
php artisan view:clear
php artisan route:clear

6. MENJALANKAN PROGRAM
----------------------
php artisan serve

7. BUKA DI BROWSER
------------------
http://127.0.0.1:8000

8. LOGIN
--------
Username : admin
Password : admin123

9. JIKA ADA ERROR
-----------------
php artisan cache:clear
php artisan view:clear
php artisan config:clear
php artisan route:clear
composer dump-autoload
php artisan serve

10. JIKA MYSQL ERROR DI XAMPP
------------------------------
- Buka XAMPP Control Panel
- Stop MySQL
- Start MySQL lagi
- Jika masih error, restart laptop

-- ============================================
-- TABEL BARANG
-- ============================================
CREATE TABLE `barang` (
  `id` bigint(20) UNSIGNED NOT NULL AUTO_INCREMENT,
  `kode_barang` varchar(20) NOT NULL,
  `nama_barang` varchar(100) NOT NULL,
  `kategori_id` bigint(20) UNSIGNED NOT NULL,
  `stok_tersedia` int(11) NOT NULL DEFAULT 0,
  `harga` decimal(15,2) NOT NULL DEFAULT 0.00,
  `status` enum('tersedia','habis','discontinue') NOT NULL DEFAULT 'tersedia',
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  `stok` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  UNIQUE KEY `barang_kode_barang_unique` (`kode_barang`),
  KEY `barang_kategori_id_foreign` (`kategori_id`),
  CONSTRAINT `barang_kategori_id_foreign` FOREIGN KEY (`kategori_id`) REFERENCES `kategori` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
